  5/**
  6//  \file GetColorPolarizedImages.cpp
  7//
  8//  \brief
  9//  This example shows how to use the functions contained in the SDK to generate some useful images with the color polarized camera.
 10//
 11//  Description: 
 12//
 13//	This example extracts a set of useful images from a color polarized camera:
 14//
 15//	- RGB and mono images for each polarization angle
 16//	- Degree of Linear Polarization (DoLP) for each colour or mono
 17//	- Angle of Linear Polarization (AoLP) for each colour or mono, also available in HSV format
 18//	- A combination of DoLP and AoLP in HSV format, where DoLP is the saturation, and AoLP the hue
 19//
 20//	Images are saved in png or RAW format.
 21
 22
 23//
 24*/

 39std::vector<std::string> ListFiles(const std::string& path)
 40{
 41	std::vector<std::string> files;
 42	WIN32_FIND_DATAA fd;
 43
 44	HANDLE h = FindFirstFileA(path.c_str(), &fd);
 45	if (h != INVALID_HANDLE_VALUE) {
 46		do {
 47			std::stringstream ss;
 48			ss << fd.cFileName;
 49			files.push_back(ss.str());
 50		} while (FindNextFileA(h, &fd));
 51		FindClose(h);
 52	}
 53	return files;
 54}

 56void CreateDirectoryIfNotExists(const std::string& dirName_in)
 57{
 58	bool dirExists;
 59	DWORD ftyp = GetFileAttributesA(dirName_in.c_str());
 60	if (ftyp == INVALID_FILE_ATTRIBUTES)
 61		dirExists = false;  //something is wrong with your path!
 62	else if (ftyp & FILE_ATTRIBUTE_DIRECTORY)
 63		dirExists = true;   // this is a directory!
 64
 65	if (!dirExists)
 66		_mkdir(dirName_in.c_str());
 67}

 69int main(int argc, char *argv[])
 70{

 72
 73	if (argc != 5)
 74	{
 75		std::cout << std::endl << std::endl << "Incorrect number of parameters, please enter the following parameters in the corresponding order:" << std::endl << std::endl;
 76		std::cout << "1 - Pixel type: Mono8, Mono10 or Mono12" << std::endl;
 77		std::cout << "2 - Input path where your images are stored" << std::endl;
 78		std::cout << "3 - Pattern to scan in the directory (i.e \"*.raw\")" << std::endl;
 79		std::cout << "4 - Output directory where the resulting images will be stored" << std::endl;
 80
 81		return -1;
 82	}
 83	// Available angles [deg_0, deg_45, deg_90, deg_135]
 84
 85	std::string pixelType = std::string(argv[1]);
 86
 87	pfPixelType m_pixelType;
 88
 89	if(pixelType == "Mono8")
 90		m_pixelType = pfPixelType::PixelMono8;
 91	else if(pixelType == "Mono10")
 92		m_pixelType = pfPixelType::PixelMono10;
 93	else if (pixelType == "Mono12")
 94		m_pixelType = pfPixelType::PixelMono12;
 95	else
 96	{
 97		std::cerr << "Unsupported pixel type " << std::endl;
 98		return -1;
 99	}

101	// Directory of the input frames
102	std::string inputPath = std::string(argv[2]);
103	if (inputPath.substr(inputPath.length() - 1) != "\\")
104		inputPath = inputPath + "\\";
105
106	std::string pattern = std::string(argv[3]);
107
108	// Output directory where to store the build hypercubes
109	std::string outputDirectory = std::string(argv[4]);
110
111	// Create the output directory if it does not exist
112	if (outputDirectory.substr(outputDirectory.length() - 1) != "\\")
113		outputDirectory = outputDirectory + "\\";
114
115
116	const uint16_t m_originalImageWidth = 2448;
117	const uint16_t m_originalImageHeight = 2048;
118	const uint16_t m_angleWidth = m_originalImageWidth/2;
119	const uint16_t m_angleHeight = m_originalImageHeight/2;
120	const uint16_t num_bytes = sizeof(uint16_t);
121

123	CreateDirectoryIfNotExists(outputDirectory);

127	std::vector<std::string> files = ListFiles(inputPath + pattern);
128
129	std::vector<std::string>::iterator it = files.begin();
130
131	if (it == files.end())
132	{
133		std::cout << "No files found in directory matching the search pattern" << std::endl;
134		return -1;
135	}

159		uint16_t* buffer_u16 = pfcAllocateImage_u16(m_originalImageWidth, m_originalImageHeight);

161		fresult = pfcReadRawImage(buffer_u16, filePath.c_str(), m_originalImageWidth, m_originalImageHeight, m_pixelType);
162		if (fresult != PFPOLSDK_NOERROR) {
163			char error[ERROR_BUFFER_LEN];
164			char errorDescription[ERROR_BUFFER_LEN];
165			pfGetErrorString(fresult, error, errorDescription);
166			std::cout << error << std::endl;
167			std::cout << errorDescription << std::endl;
168			pfcFreeImage_u16(buffer_u16, m_originalImageWidth, m_originalImageHeight);
169			return -1;
170		}

172		///// ppp_colorimages stores a mono image for each color and angle with the format ppp_result[angle][color], where:
173		// angles are [0 - 90deg, 1 - 45deg, 2 - 135deg, 3 - 0deg]
174		// colors are [0 - Red, 1 - Green, 2 - Blue]
175		/////////////
176
177		// Allocate polarized image
178		PolImgPtr_t pppp_colorimages;
179		pfcAllocatePolarizedImage(&pppp_colorimages, m_angleWidth, m_angleHeight);
180		// Load input raw image buffer to polarized image
181		fresult = pfcLoadPolarizedRGBImage(pppp_colorimages, buffer_u16, m_originalImageWidth, m_originalImageHeight, m_angleWidth, m_angleHeight, m_pixelType);
182		if (fresult != PFPOLSDK_NOERROR) {
183			char error[ERROR_BUFFER_LEN];
184			char errorDescription[ERROR_BUFFER_LEN];
185			pfGetErrorString(fresult, error, errorDescription);
186			std::cout << error << std::endl;
187			std::cout << errorDescription << std::endl;
188			free(buffer_u16);
189			pfcFreeColorPolarizedImage(pppp_colorimages);
190			return -1;
191		}

193		///// Using pppp_colorimages, pppp_monoresult stores a mono image for each angle with the format ppp_monoresult[angle][color], where
194		// angles are [0 - 90deg, 1 - 45deg, 2 - 135deg, 3 - 0deg]
195		// colors are [0 - Mono]
196		/////////////
197		PolImgPtr_t pppp_monoimages;
198		fresult = pfcGeneratePolarizedMonoImage(&pppp_monoimages, pppp_colorimages, m_originalImageWidth, m_originalImageHeight, m_angleWidth, m_angleHeight);
199		if (fresult != PFPOLSDK_NOERROR) {
200			char error[ERROR_BUFFER_LEN];
201			char errorDescription[ERROR_BUFFER_LEN];
202			pfGetErrorString(fresult, error, errorDescription);
203			std::cout << error << std::endl;
204			std::cout << errorDescription << std::endl;

207			pfcFreeColorPolarizedImage(pppp_monoimages);
